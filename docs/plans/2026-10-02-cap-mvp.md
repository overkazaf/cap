**Goal:** Build a Go-based MITM proxy tool that produces LLM-ready output, supports static analysis context correlation, generates multi-language request scripts, and provides one-command Android traffic capture — CLI-first with embedded Web GUI.

**Architecture:** Single Go binary with 5 decoupled packages behind internal interfaces: `proxy` (MITM core), `store` (SQLite persistence), `export` (code generation + agent output), `context` (static analysis mapping), `android` (ADB-based device setup). **Tech Stack:** Go 1.27, goproxy (MITM), cobra (CLI), SQLite via modernc.org/sqlite (pure Go), gorilla/websocket, embed (static assets), text/template (codegen), ADB (Android Debug Bridge via shell exec)

```go
package proxy

import (
	"bytes"
	"io"
	"net"
	"net/http"
	"strings"
	"sync"
	"sync/atomic"
	"time"

	"github.com/elazarl/goproxy"
	"github.com/user/cap/internal/types"
)

type Options struct {
	Addr   string
	OnFlow func(*types.Flow)
}

type Proxy struct {
	opts     Options
	gp       *goproxy.ProxyHttpServer
	listener net.Listener
	server   *http.Server
	running  atomic.Bool
	wg       sync.WaitGroup
	idSeq    atomic.Int64
}

func New(opts Options) (*Proxy, error) {
	gp := goproxy.NewProxyHttpServer()
	gp.Verbose = false

	ln, err := net.Listen("tcp", opts.Addr)
	if err != nil {
		return nil, err
	}

	p := &Proxy{
		opts:     opts,
		gp:       gp,
		listener: ln,
		server:   &http.Server{Handler: gp},
	}

	type flowMeta struct {
		start time.Time
		reqBody []byte
	}
	var mu sync.Mutex
	pending := make(map[*http.Request]*flowMeta)

	gp.OnRequest().DoFunc(func(req *http.Request, ctx *goproxy.ProxyCtx) (*http.Request, *http.Response) {
		meta := &flowMeta{start: time.Now()}
		if req.Body != nil {
			body, _ := io.ReadAll(req.Body)
			req.Body = io.NopCloser(bytes.NewReader(body))
			meta.reqBody = body
		}
		mu.Lock()
		pending[req] = meta
		mu.Unlock()
		return req, nil
	})

	gp.OnResponse().DoFunc(func(resp *http.Response, ctx *goproxy.ProxyCtx) *http.Response {
		if resp == nil || ctx.Req == nil {
			return resp
		}
		mu.Lock()
		meta, ok := pending[ctx.Req]
		if ok {
			delete(pending, ctx.Req)
		}
		mu.Unlock()
		if !ok {
			return resp
		}

		var respBody []byte
		if resp.Body != nil {
			respBody, _ = io.ReadAll(resp.Body)
			resp.Body = io.NopCloser(bytes.NewReader(respBody))
		}

		seq := p.idSeq.Add(1)
		flow := &types.Flow{
			ID:          fmt.Sprintf("f%d", seq),
			Timestamp:   meta.start,
			Method:      ctx.Req.Method,
			URL:         ctx.Req.URL.String(),
			Host:        ctx.Req.URL.Host,
			Path:        ctx.Req.URL.Path,
			ReqHeaders:  flattenHeaders(ctx.Req.Header),
			ReqBody:     meta.reqBody,
			ReqBodyType: ctx.Req.Header.Get("Content-Type"),
			Status:      resp.StatusCode,
			RespHeaders: flattenHeaders(resp.Header),
			RespBody:    respBody,
			RespBodyType: resp.Header.Get("Content-Type"),
			LatencyMs:   time.Since(meta.start).Milliseconds(),
		}

		if opts.OnFlow != nil {
			opts.OnFlow(flow)
		}
		return resp
	})

	return p, nil
}

func (p *Proxy) Start() error {
	p.running.Store(true)
	return p.server.Serve(p.listener)
}

func (p *Proxy) Stop() error {
	p.running.Store(false)
	return p.server.Close()
}

func (p *Proxy) Addr() string {
	return p.listener.Addr().String()
}

func flattenHeaders(h http.Header) map[string]string {
	m := make(map[string]string, len(h))
	for k, v := range h {
		m[k] = strings.Join(v, "; ")
	}
	return m
}
```

```go
package proxy

import (
	"crypto/ecdsa"
	"crypto/elliptic"
	"crypto/rand"
	"crypto/x509"
	"crypto/x509/pkix"
	"encoding/pem"
	"math/big"
	"os"
	"path/filepath"
	"time"
)

func EnsureCA(dir string) (certFile, keyFile string, err error) {
	certFile = filepath.Join(dir, "cap-ca.crt")
	keyFile = filepath.Join(dir, "cap-ca.key")

	if _, err := os.Stat(certFile); err == nil {
		return certFile, keyFile, nil
	}

	key, err := ecdsa.GenerateKey(elliptic.P256(), rand.Reader)
	if err != nil {
		return "", "", err
	}

	serial, _ := rand.Int(rand.Reader, new(big.Int).Lsh(big.NewInt(1), 128))
	tmpl := &x509.Certificate{
		SerialNumber: serial,
		Subject:      pkix.Name{CommonName: "Cap MITM CA", Organization: []string{"cap"}},
		NotBefore:    time.Now().Add(-time.Hour),
		NotAfter:     time.Now().Add(10 * 365 * 24 * time.Hour),
		KeyUsage:     x509.KeyUsageCertSign | x509.KeyUsageCRLSign,
		IsCA:         true,
		BasicConstraintsValid: true,
	}

	certDER, err := x509.CreateCertificate(rand.Reader, tmpl, tmpl, &key.PublicKey, key)
	if err != nil {
		return "", "", err
	}

	os.MkdirAll(dir, 0700)

	cf, _ := os.Create(certFile)
	pem.Encode(cf, &pem.Block{Type: "CERTIFICATE", Bytes: certDER})
	cf.Close()

	keyDER, _ := x509.MarshalECPrivateKey(key)
	kf, _ := os.OpenFile(keyFile, os.O_CREATE|os.O_WRONLY, 0600)
	pem.Encode(kf, &pem.Block{Type: "EC PRIVATE KEY", Bytes: keyDER})
	kf.Close()

	return certFile, keyFile, nil
}
```

```go
package proxy

import "strings"

var noiseHeaders = map[string]bool{
	"accept-encoding":           true,
	"connection":                true,
	"proxy-connection":          true,
	"proxy-authorization":       true,
	"te":                        true,
	"trailer":                   true,
	"transfer-encoding":         true,
	"upgrade":                   true,
	"keep-alive":                true,
	"x-forwarded-for":           true,
	"x-forwarded-proto":         true,
}

func StripNoiseHeaders(headers map[string]string) map[string]string {
	clean := make(map[string]string, len(headers))
	for k, v := range headers {
		if !noiseHeaders[strings.ToLower(k)] {
			clean[k] = v
		}
	}
	return clean
}
```

```go
package sqlite

import (
	"database/sql"
	"encoding/json"
	"fmt"
	"strings"
	"time"

	_ "modernc.org/sqlite"

	"github.com/user/cap/internal/store"
	"github.com/user/cap/internal/types"
)

var _ store.Store = (*SQLiteStore)(nil)

type SQLiteStore struct {
	db *sql.DB
}

func New(dsn string) (*SQLiteStore, error) {
	db, err := sql.Open("sqlite", dsn)
	if err != nil {
		return nil, err
	}
	db.Exec("PRAGMA journal_mode=WAL")
	db.Exec("PRAGMA busy_timeout=5000")

	_, err = db.Exec(`CREATE TABLE IF NOT EXISTS flows (
		id TEXT PRIMARY KEY,
		timestamp TEXT NOT NULL,
		method TEXT NOT NULL,
		url TEXT NOT NULL,
		host TEXT NOT NULL DEFAULT '',
		path TEXT NOT NULL DEFAULT '',
		req_headers TEXT DEFAULT '{}',
		req_body BLOB,
		req_body_type TEXT DEFAULT '',
		status INTEGER NOT NULL DEFAULT 0,
		resp_headers TEXT DEFAULT '{}',
		resp_body BLOB,
		resp_body_type TEXT DEFAULT '',
		latency_ms INTEGER DEFAULT 0,
		tags TEXT DEFAULT '[]',
		sign_params TEXT DEFAULT '[]',
		source_ref TEXT
	)`)
	if err != nil {
		return nil, err
	}
	db.Exec("CREATE INDEX IF NOT EXISTS idx_host ON flows(host)")
	db.Exec("CREATE INDEX IF NOT EXISTS idx_method ON flows(method)")
	db.Exec("CREATE INDEX IF NOT EXISTS idx_status ON flows(status)")

	return &SQLiteStore{db: db}, nil
}

func (s *SQLiteStore) SaveFlow(f *types.Flow) error {
	reqH, _ := json.Marshal(f.ReqHeaders)
	respH, _ := json.Marshal(f.RespHeaders)
	tags, _ := json.Marshal(f.Tags)
	signP, _ := json.Marshal(f.SignParams)
	var srcRef []byte
	if f.SourceRef != nil {
		srcRef, _ = json.Marshal(f.SourceRef)
	}
	_, err := s.db.Exec(`INSERT OR REPLACE INTO flows
		(id, timestamp, method, url, host, path, req_headers, req_body, req_body_type,
		 status, resp_headers, resp_body, resp_body_type, latency_ms, tags, sign_params, source_ref)
		VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
		f.ID, f.Timestamp.Format(time.RFC3339Nano), f.Method, f.URL, f.Host, f.Path,
		string(reqH), f.ReqBody, f.ReqBodyType,
		f.Status, string(respH), f.RespBody, f.RespBodyType,
		f.LatencyMs, string(tags), string(signP), string(srcRef),
	)
	return err
}

func (s *SQLiteStore) GetFlow(id string) (*types.Flow, error) {
	row := s.db.QueryRow("SELECT * FROM flows WHERE id = ?", id)
	return scanFlow(row)
}

func (s *SQLiteStore) ListFlows(filter types.FlowFilter) ([]*types.Flow, error) {
	var conditions []string
	var args []any

	if filter.Host != "" {
		conditions = append(conditions, "host = ?")
		args = append(args, filter.Host)
	}
	if filter.Method != "" {
		conditions = append(conditions, "method = ?")
		args = append(args, filter.Method)
	}
	if filter.Path != "" {
		conditions = append(conditions, "path LIKE ?")
		args = append(args, "%"+filter.Path+"%")
	}
	if filter.StatusFrom > 0 {
		conditions = append(conditions, "status >= ?")
		args = append(args, filter.StatusFrom)
	}
	if filter.StatusTo > 0 {
		conditions = append(conditions, "status <= ?")
		args = append(args, filter.StatusTo)
	}
	if filter.Tag != "" {
		conditions = append(conditions, "tags LIKE ?")
		args = append(args, fmt.Sprintf("%%\"%s\"%%", filter.Tag))
	}
	if filter.Search != "" {
		conditions = append(conditions, "(url LIKE ? OR CAST(req_body AS TEXT) LIKE ?)")
		args = append(args, "%"+filter.Search+"%", "%"+filter.Search+"%")
	}

	query := "SELECT * FROM flows"
	if len(conditions) > 0 {
		query += " WHERE " + strings.Join(conditions, " AND ")
	}
	query += " ORDER BY timestamp DESC"

	if filter.Limit > 0 {
		query += fmt.Sprintf(" LIMIT %d", filter.Limit)
	}
	if filter.Offset > 0 {
		query += fmt.Sprintf(" OFFSET %d", filter.Offset)
	}

	rows, err := s.db.Query(query, args...)
	if err != nil {
		return nil, err
	}
	defer rows.Close()

	var flows []*types.Flow
	for rows.Next() {
		f, err := scanFlowRows(rows)
		if err != nil {
			return nil, err
		}
		flows = append(flows, f)
	}
	return flows, nil
}

func (s *SQLiteStore) DeleteFlow(id string) error {

type scannable interface {
	Scan(dest ...any) error
}

func scanFlow(row *sql.Row) (*types.Flow, error) {
	var f types.Flow
```

```go
package codegen

import (
	"bytes"
	"embed"
	"encoding/json"
	"fmt"
	"sort"
	"strings"
	"text/template"

	"github.com/user/cap/internal/types"

//go:embed templates/*.tmpl
var tmplFS embed.FS

var templates *template.Template

func init() {
	funcMap := template.FuncMap{
		"jsonPretty": func(b []byte) string {
			var buf bytes.Buffer
			if json.Indent(&buf, b, "", "  ") == nil {
				return buf.String()
			}
			return string(b)
		},
		"escapeShell": func(s string) string {
			return strings.ReplaceAll(s, "'", "'\\''")
		},
		"sortedHeaders": func(h map[string]string) []struct{ K, V string } {
			keys := make([]string, 0, len(h))
			for k := range h {
				keys = append(keys, k)
			}
			sort.Strings(keys)
			result := make([]struct{ K, V string }, len(keys))
			for i, k := range keys {
				result[i] = struct{ K, V string }{k, h[k]}
			}
			return result
		},
		"hasBody": func(f *types.Flow) bool {
			return len(f.ReqBody) > 0
		},
		"isJSON": func(ct string) bool {
			return strings.Contains(ct, "json")
		},
	}
	templates = template.Must(template.New("").Funcs(funcMap).ParseFS(tmplFS, "templates/*.tmpl"))
}

func Languages() []string {
	return []string{"curl", "python", "go", "java", "js"}
}

func Generate(flow *types.Flow, lang string) (string, error) {
	tmplName := lang + ".tmpl"
	t := templates.Lookup(tmplName)
	if t == nil {
		return "", fmt.Errorf("unknown language: %s", lang)
	}
	var buf bytes.Buffer
	if err := t.Execute(&buf, flow); err != nil {
		return "", err
	}
	return buf.String(), nil
}
```

```go
package agent

import (
	"encoding/json"
	"net/url"
	"regexp"
	"strings"

	"github.com/user/cap/internal/types"
)

var signKeyPatterns = []*regexp.Regexp{
	regexp.MustCompile(`(?i)^sign(ature)?$`),
	regexp.MustCompile(`(?i)^(time_?stamp|ts|_t)$`),
	regexp.MustCompile(`(?i)^nonce$`),
	regexp.MustCompile(`(?i)^token$`),
	regexp.MustCompile(`(?i)^hmac$`),
	regexp.MustCompile(`(?i)^hash$`),
	regexp.MustCompile(`(?i)^digest$`),
	regexp.MustCompile(`(?i)^app_?key$`),
	regexp.MustCompile(`(?i)^secret$`),
}

func DetectSignParams(flow *types.Flow) []string {
	seen := make(map[string]bool)

	if u, err := url.Parse(flow.URL); err == nil {
		for key := range u.Query() {
			if isSignKey(key) {
				seen[key] = true
			}
		}
	}

	if len(flow.ReqBody) > 0 {
		var bodyMap map[string]any
		if json.Unmarshal(flow.ReqBody, &bodyMap) == nil {
			for key := range bodyMap {
				if isSignKey(key) {
					seen[key] = true
				}
			}
		}

		if vals, err := url.ParseQuery(string(flow.ReqBody)); err == nil {
			for key := range vals {
				if isSignKey(key) {
					seen[key] = true
				}
			}
		}
	}

	result := make([]string, 0, len(seen))
	for k := range seen {
		result = append(result, k)
	}
	return result
}

func isSignKey(key string) bool {
	k := strings.ToLower(key)
	for _, p := range signKeyPatterns {
		if p.MatchString(k) {
			return true
		}
	}
	return false
}
```

```go
package agent

import (
	"crypto/sha256"
	"encoding/json"
	"fmt"
	"strings"

	"github.com/user/cap/internal/proxy"
	"github.com/user/cap/internal/types"
)

type FormatOptions struct {
	MaxBodySize int
	StripNoise  bool
	IncludeMeta bool
}

type compactFlow struct {
	ID          string            `json:"id"`
	TS          int64             `json:"ts"`
	Method      string            `json:"method"`
	URL         string            `json:"url"`
	ReqHeaders  map[string]string `json:"req_headers,omitempty"`
	ReqBody     any               `json:"req_body,omitempty"`
	Status      int               `json:"status"`
	RespBody    any               `json:"resp_body,omitempty"`
	LatencyMs   int64             `json:"latency_ms"`
	Tags        []string          `json:"tags,omitempty"`
	SignParams  []string          `json:"sign_params,omitempty"`
	SourceRef   *types.SourceRef  `json:"source_ref,omitempty"`
	BodyHash    string            `json:"body_hash,omitempty"`
}

func Format(flows []*types.Flow, opts FormatOptions) ([]byte, error) {
	if opts.MaxBodySize == 0 {
		opts.MaxBodySize = 4096
	}
	var lines []string
	for _, f := range flows {
		cf := compactFlow{
			ID:         f.ID,
			TS:         f.Timestamp.Unix(),
			Method:     f.Method,
			URL:        f.URL,
			ReqHeaders: proxy.StripNoiseHeaders(f.ReqHeaders),
			Status:     f.Status,
			LatencyMs:  f.LatencyMs,
			Tags:       f.Tags,
			SignParams: f.SignParams,
			SourceRef:  f.SourceRef,
		}

		if sp := DetectSignParams(f); len(sp) > 0 && len(cf.SignParams) == 0 {
			cf.SignParams = sp
		}

		cf.ReqBody = compactBody(f.ReqBody, f.ReqBodyType, opts.MaxBodySize)
		rb, hash := compactBodyWithHash(f.RespBody, f.RespBodyType, opts.MaxBodySize)
		cf.RespBody = rb
		cf.BodyHash = hash

		b, err := json.Marshal(cf)
		if err != nil {
			return nil, err
		}
		lines = append(lines, string(b))
	}
	return []byte(strings.Join(lines, "\n") + "\n"), nil
}

func compactBody(body []byte, contentType string, maxSize int) any {
	if len(body) == 0 {
		return nil
	}
	if strings.Contains(contentType, "json") {
		var v any
		if json.Unmarshal(body, &v) == nil {
			b, _ := json.Marshal(v)
			if len(b) <= maxSize {
				return v
			}
			return truncateJSON(v, maxSize)
		}
	}
	s := string(body)
	if len(s) > maxSize {
		return s[:maxSize] + "...[truncated]"
	}
	return s
}

func compactBodyWithHash(body []byte, contentType string, maxSize int) (any, string) {
	if len(body) == 0 {
		return nil, ""
	}
	var hash string
	if len(body) > maxSize {
		h := sha256.Sum256(body)
		hash = fmt.Sprintf("sha256:%x", h[:8])
	}
	return compactBody(body, contentType, maxSize), hash
}

func truncateJSON(v any, maxSize int) any {
	b, _ := json.Marshal(v)
	if len(b) <= maxSize {
		return v
	}
	return string(b[:maxSize]) + "...[truncated]"
}
```

**Design reference — HTTP Toolkit's approach:**
- Detects connected devices via `adb devices`
- Sets Wi-Fi proxy via `adb shell settings put global http_proxy <host>:<port>`
- On rooted devices: pushes CA cert to system trust store (`/system/etc/security/cacerts/`)
- On non-root: installs CA cert as user cert via `adb shell am start` intent
- Monitors device connectivity and auto-reconnects

Cap's approach (simpler, CLI-first):
- `cap android connect` — auto-detect device, set Wi-Fi proxy, push CA cert
- `cap android disconnect` — restore original proxy settings, remove cert
- `cap android status` — show device state and proxy config
- Works with both USB and Wi-Fi ADB (`adb connect <ip>:5555`)

```go
		if line == "" || strings.HasPrefix(line, "List of") || strings.HasPrefix(line, "*") {
			continue
		}
		parts := strings.Fields(line)
		if len(parts) >= 2 && parts[1] == "device" {
			devices = append(devices, Device{Serial: parts[0], State: parts[1]})
		}
	}
	return devices
}

func ParseProxySetting(setting string) (host, port string) {
	setting = strings.TrimSpace(setting)
	if setting == "" || setting == "null" || setting == ":0" {
		return "", ""
	}
	if idx := strings.LastIndex(setting, ":"); idx > 0 {
		return setting[:idx], setting[idx+1:]
	}
	return setting, ""
}

func ListDevices() ([]Device, error) {
	out, err := adbCmd("", "devices")
	if err != nil {
		return nil, fmt.Errorf("adb devices: %w", err)
	}
	return ParseDeviceList(out), nil
}

func GetDeviceSerial(preferred string) (string, error) {
	devices, err := ListDevices()
	if err != nil {
		return "", err
	}
	if len(devices) == 0 {
		return "", fmt.Errorf("no Android device connected (run 'adb devices' to check)")
	}
	if preferred != "" {
		for _, d := range devices {
			if d.Serial == preferred {
				return d.Serial, nil
			}
		}
		return "", fmt.Errorf("device %q not found, available: %v", preferred, devices)
	}
	return devices[0].Serial, nil
}

func adbCmd(serial string, args ...string) (string, error) {
	cmdArgs := []string{}
	if serial != "" {
		cmdArgs = append(cmdArgs, "-s", serial)
	}
	cmdArgs = append(cmdArgs, args...)
	cmd := exec.Command("adb", cmdArgs...)
	out, err := cmd.CombinedOutput()
	return strings.TrimSpace(string(out)), err
}

func adbShell(serial string, shellCmd string) (string, error) {
	return adbCmd(serial, "shell", shellCmd)
}

func WaitForDevice(serial string, timeout time.Duration) error {
	args := []string{}
	if serial != "" {
		args = append(args, "-s", serial)
	}
	args = append(args, "wait-for-device")
	cmd := exec.Command("adb", args...)
	done := make(chan error, 1)
	go func() { done <- cmd.Run() }()
	select {
	case err := <-done:
		return err
	case <-time.After(timeout):
		cmd.Process.Kill()
		return fmt.Errorf("timeout waiting for device %s", serial)
	}
}
```

```go
import (
	"crypto/x509"
	"encoding/pem"
	"fmt"
	"os"

func CertHashFilename(certPath string) string {
	data, err := os.ReadFile(certPath)
	if err != nil {
		return ""
	}
	block, _ := pem.Decode(data)
	if block == nil {
		return ""
	}
	cert, err := x509.ParseCertificate(block.Bytes)
	if err != nil {
		return ""
	}
	// Android uses subject hash in OpenSSL format
	// simplified: use first 8 hex chars of subject hash
	h := fmt.Sprintf("%08x", subjectNameHash(cert))
	return h + ".0"
}

func subjectNameHash(cert *x509.Certificate) uint32 {
	// OpenSSL X509_NAME_hash — simplified MD5-based approach
	raw := cert.RawSubject
	sum := uint32(0)
	for _, b := range raw {
		sum = sum*31 + uint32(b)
	}
	return sum
}

func installCACert(serial, certPath string) error {
	if isRooted(serial) {
		return installCACertRooted(serial, certPath)
	}
	return installCACertUser(serial, certPath)
}

func installCACertRooted(serial, certPath string) error {
	hashName := CertHashFilename(certPath)
	if hashName == "" {
		hashName = "cap-ca.0"
	}
	remotePath := "/data/local/tmp/" + hashName
	systemPath := "/system/etc/security/cacerts/" + hashName

	if _, err := adbCmd(serial, "push", certPath, remotePath); err != nil {
		return fmt.Errorf("push cert: %w", err)
	}

	cmds := []string{
		"su -c mount -o remount,rw /system",
		fmt.Sprintf("su -c cp %s %s", remotePath, systemPath),
		fmt.Sprintf("su -c chmod 644 %s", systemPath),
		"su -c mount -o remount,ro /system",
		fmt.Sprintf("rm %s", remotePath),
	}
	for _, cmd := range cmds {
		if _, err := adbShell(serial, cmd); err != nil {
			return fmt.Errorf("cmd %q: %w", cmd, err)
		}
	}
	return nil
}

func installCACertUser(serial, certPath string) error {
	remotePath := "/data/local/tmp/cap-ca.crt"
	if _, err := adbCmd(serial, "push", certPath, remotePath); err != nil {
		return fmt.Errorf("push cert: %w", err)
	}

	// open cert installer via intent
	intent := fmt.Sprintf(
		"am start -a android.intent.action.VIEW -t application/x-x509-ca-cert -d file://%s",
		remotePath,
	)
	_, err := adbShell(serial, intent)
	if err != nil {
		// fallback: settings screen
		adbShell(serial, "am start -a android.settings.SECURITY_SETTINGS")
		fmt.Printf("cert pushed to %s — install manually from Settings > Security\n", remotePath)
	}
	return err
}

func removeCACert(serial string, certPath string) error {
	if !isRooted(serial) {
		fmt.Println("non-root: remove the CA cert manually from Settings > Security > Trusted Credentials > User")
		return nil
	}
	hashName := CertHashFilename(certPath)
	if hashName == "" {
		return nil
	}
	systemPath := "/system/etc/security/cacerts/" + hashName
	cmds := []string{
		"su -c mount -o remount,rw /system",
		fmt.Sprintf("su -c rm -f %s", systemPath),
		"su -c mount -o remount,ro /system",
	}
	for _, cmd := range cmds {
		adbShell(serial, cmd)
	}
	return nil
}
```

```go
func ScanJadxOutput(dir string) ([]Mapping, error) {
	var mappings []Mapping

	err := filepath.Walk(dir, func(path string, info os.FileInfo, err error) error {
		if err != nil || info.IsDir() {
			return err
		}
		if !strings.HasSuffix(path, ".java") {
			return nil
		}

		relPath, _ := filepath.Rel(dir, path)
		ms, err := parseJavaFile(path, relPath)
		if err != nil {
			return nil
		}
		mappings = append(mappings, ms...)
		return nil
	})
	return mappings, err
}

func parseJavaFile(absPath, relPath string) ([]Mapping, error) {
	f, err := os.Open(absPath)
	if err != nil {
		return nil, err
	}
	defer f.Close()

	var mappings []Mapping
	scanner := bufio.NewScanner(f)
	lineNum := 0
	var pendingMethod, pendingURL string
	var pendingLine int

	for scanner.Scan() {
		lineNum++
		line := scanner.Text()

		if m := retrofitAnnotation.FindStringSubmatch(line); len(m) == 3 {
			pendingMethod = m[1]
			pendingURL = m[2]
			pendingLine = lineNum
			continue
		}

		if pendingMethod != "" {
			if m := methodDeclAfterAnnotation.FindStringSubmatch(line); len(m) == 2 {
				mappings = append(mappings, Mapping{
					URLPattern: pendingMethod + " " + pendingURL,
					Source: types.SourceRef{
						File:   relPath,
						Method: m[1],
						Line:   pendingLine,
					},
				})
			}
			pendingMethod = ""
		}
	}
	return mappings, nil
}
```

**Step 5: Implement matcher.go**

```go
package context

import (
	"strings"

	"github.com/user/cap/internal/types"
)

type Matcher struct {
	mappings []Mapping
}

func NewMatcher(mappings []Mapping) *Matcher {
	return &Matcher{mappings: mappings}
}

func (m *Matcher) Match(flow *types.Flow) *types.SourceRef {
	if m == nil || len(m.mappings) == 0 {
		return nil
	}
	for _, mapping := range m.mappings {
		parts := strings.SplitN(mapping.URLPattern, " ", 2)
		if len(parts) != 2 {
			continue
		}
		method, pattern := parts[0], parts[1]

		if !strings.EqualFold(flow.Method, method) {
			continue
		}
		if matchPath(flow.Path, pattern) {
			ref := mapping.Source
			return &ref
		}
	}
	return nil
}

func matchPath(actual, pattern string) bool {
	if actual == pattern {
		return true
	}
	// handle path params like /user/{id}
	aParts := strings.Split(strings.Trim(actual, "/"), "/")
	pParts := strings.Split(strings.Trim(pattern, "/"), "/")
	if len(aParts) != len(pParts) {
		return false
	}
	for i := range pParts {
		if strings.HasPrefix(pParts[i], "{") && strings.HasSuffix(pParts[i], "}") {
			continue
		}
		if aParts[i] != pParts[i] {
			return false
		}
	}
	return true
}
```

## MVP Definition

**What ships:** `cap start`, `cap flows`, `cap export`, `cap android` — a single Go binary that:
1. Runs as MITM proxy capturing HTTP/HTTPS traffic
2. Stores flows in SQLite
3. Lists/filters/searches captured flows
4. Exports any flow as cURL/Python/Go/Java/JS
5. Exports all flows as compact JSONL for LLM consumption
6. Auto-detects sign/timestamp/nonce parameters
7. Scans jadx output to map URLs to source code
8. Matches captured flows to source code references
9. One-command Android device proxy setup via ADB (`cap android connect`)
10. Auto-detects host IP, pushes CA cert, supports root and non-root devices

**What doesn't ship in MVP:** Web GUI, Frida hook import, session grouping, VPN-based capture.
